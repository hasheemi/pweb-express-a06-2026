// --- ENUMS ---


enum Role {
  TANK
  FIGHTER
  ASSASSIN
  MAGE
  MARKSMAN
  SUPPORT
}


enum Lane {
  EXP
  GOLD
  MID
  JUNGLE
  ROAM
}


enum Rarity {
  COMMON
  EXCEPTIONAL
  DELUXE
  EXQUISITE
  GRAND
  SUPREME
}


enum Currency {
  BP
  DIAMOND
}


// --- MODELS ---


model Hero {
  id           Int          @id @default(autoincrement())
  name         String       @unique // Dijelaskan di poin: "Kolom unik: Hero.name"
  role         Role
  lane         Lane
  releaseYear  Int
  priceBP      Int
  priceDiamond Int


  skins        Skin[]
  playerHeros  PlayerHero[]
}


model Skin {
  id           Int    @id @default(autoincrement())
  name         String
  rarity       Rarity
  priceDiamond Int
  heroId       Int


  // Relasi ke Hero dengan onDelete Cascade
  // "Pasang onDelete: Cascade pada relasi Skin -> Hero"
  hero         Hero   @relation(fields: [heroId], references: [id], onDelete: Cascade)
}


model Player {
  id        Int      @id @default(autoincrement())
  nickname  String   @unique // Diambil dari penjelasan: "Player.nickname" unik
  bp        Int      @default(10000)
  diamond   Int      @default(0)
  createdAt DateTime @default(now())


  playerHeros PlayerHero[]
}


model PlayerHero {
  id        Int      @id @default(autoincrement())
  playerId  Int
  heroId    Int
  currency  Currency
  pricePaid Int


  // Relasi ke Player dengan onDelete Cascade
  // "Pasang onDelete: Cascade pada relasi ... PlayerHero -> Player"
  player    Player   @relation(fields: [playerId], references: [id], onDelete: Cascade)
  
  // Relasi ke Hero
  hero      Hero     @relation(fields: [heroId], references: [id])


  // Composite Unique Constraint
  // "pasangan playerId + heroId di PlayerHero"
  @@unique([playerId, heroId])
}
